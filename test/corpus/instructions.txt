==================
Adjacent instruction substitutions separated by spaces
==================
> instructions: Use <first> <last> now.
> instructions: <first> <`last`>

---

(source_file
  (system_directive
    (system_expr
      (text_segment)
      (system_angle_segment)
      (system_angle_segment)
      (text_segment)))
  (system_directive
    (system_expr
      (system_angle_segment)
      (system_angle_segment))))
